cmake_minimum_required(VERSION 3.22)
project(LightTouchPortable VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
option(LIGHTTOUCH_ENABLE_NETWORK "Shared HTTP, legacy TLS and Wayback backend" OFF)
if(LIGHTTOUCH_ENABLE_NETWORK)
 find_package(OpenSSL 3 REQUIRED)
 add_library(lighttouch_network STATIC network/Network.cpp)
 set_target_properties(lighttouch_network PROPERTIES POSITION_INDEPENDENT_CODE ON)
 target_include_directories(lighttouch_network PUBLIC network)
 target_link_libraries(lighttouch_network PUBLIC OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
 if(WIN32)
  target_link_libraries(lighttouch_network PUBLIC ws2_32)
 endif()
 target_compile_options(lighttouch_network PRIVATE -Wall -Wextra -Werror)
endif()
set(LIGHTTOUCH_QEMU_SOURCE "" CACHE PATH "qemu-ios source providing versioned portable headers")
if(NOT EXISTS "${LIGHTTOUCH_QEMU_SOURCE}/contrib/ios-app/qemu-ios-ui.h")
 message(FATAL_ERROR "Set LIGHTTOUCH_QEMU_SOURCE to the portable qemu-ios source")
endif()
add_library(lighttouch_runtime STATIC src/Storage.cpp src/Runtime.cpp src/GuestPackage.cpp src/Library.cpp src/Screenshot.cpp)
set_target_properties(lighttouch_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(lighttouch_runtime PUBLIC include PRIVATE "${LIGHTTOUCH_QEMU_SOURCE}/contrib/ios-app" "${LIGHTTOUCH_QEMU_SOURCE}/contrib/macos-app")
target_link_libraries(lighttouch_runtime PUBLIC ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(lighttouch_runtime PRIVATE -Wall -Wextra -Werror)
# Full prepared-base integrity reads also run in diagnostic builds. Optimize
# both translation units that emit inline Sha256 methods, and their contract
# test, consistently so weak/COMDAT selection cannot mix optimized and baseline
# copies. Preserve assertions, hardening/debug flags and release build policies.
# Clang's MSVC frontend and AppleClang deliberately retain their existing flags.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR
   (CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND
    CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU"))
 set(lighttouch_sha_debug_optimization "$<$<OR:$<CONFIG:Debug>,$<STREQUAL:$<CONFIG>,>>:-O2>")
 set_property(SOURCE src/Storage.cpp src/GuestPackage.cpp tests/Contracts.cpp
              APPEND PROPERTY COMPILE_OPTIONS "${lighttouch_sha_debug_optimization}")
endif()
option(LIGHTTOUCH_ENABLE_DEBUG_QMP "Explicit private diagnostic QMP hook; never enabled in release" OFF)
if(LIGHTTOUCH_ENABLE_DEBUG_QMP)
 target_compile_definitions(lighttouch_runtime PRIVATE LIGHTTOUCH_ENABLE_DEBUG_QMP=1)
endif()
option(LIGHTTOUCH_BUILD_TESTS "Synthetic ABI/storage regression tests; no guest claims" OFF)
if(LIGHTTOUCH_BUILD_TESTS)
 enable_testing()
 if(LIGHTTOUCH_ENABLE_NETWORK AND NOT WIN32)
  add_executable(lighttouch_network_test tests/network/NetworkTests.cpp)
  target_link_libraries(lighttouch_network_test PRIVATE lighttouch_network)
  target_compile_options(lighttouch_network_test PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -UNDEBUG)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
   # Android app storage may forbid hard links. Exercise identity persistence
   # under the same restriction on the host, rather than requiring a phone.
   target_compile_definitions(lighttouch_network_test PRIVATE LT_TEST_NO_HARD_LINKS=1)
   target_link_options(lighttouch_network_test PRIVATE -Wl,--wrap=link -Wl,--wrap=linkat)
   add_executable(lighttouch_network_live_test tests/network/LiveConfiguration.cpp)
   target_link_libraries(lighttouch_network_live_test PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
   target_compile_options(lighttouch_network_live_test PRIVATE -Wall -Wextra -Werror -UNDEBUG "$<$<CXX_COMPILER_ID:GNU>:-Wno-subobject-linkage>")
   target_link_options(lighttouch_network_live_test PRIVATE -Wl,--wrap=getaddrinfo -Wl,--wrap=connect)
   add_test(NAME lighttouch_network_live COMMAND lighttouch_network_live_test)
   set_tests_properties(lighttouch_network_live PROPERTIES TIMEOUT 30)
  endif()
  add_test(NAME lighttouch_network COMMAND lighttouch_network_test)
 endif()
 add_executable(lighttouch_contract_test tests/Contracts.cpp)
 target_link_libraries(lighttouch_contract_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_contract_test PRIVATE -UNDEBUG)
 add_test(NAME lighttouch_contract COMMAND lighttouch_contract_test)
 # Public C ABI only: do not emit test-local inline SHA methods that could
 # replace the archive's implementation during this integrity regression.
 add_executable(lighttouch_file_hash_test tests/FileHash.cpp)
 target_link_libraries(lighttouch_file_hash_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_file_hash_test PRIVATE -UNDEBUG)
 add_test(NAME lighttouch_file_hash COMMAND lighttouch_file_hash_test)
 add_executable(lighttouch_archive_import_test tests/ArchiveImport.cpp)
 target_link_libraries(lighttouch_archive_import_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_archive_import_test PRIVATE -UNDEBUG)
 set_property(SOURCE tests/ArchiveImport.cpp APPEND PROPERTY COMPILE_OPTIONS "${lighttouch_sha_debug_optimization}")
 add_test(NAME lighttouch_archive_import COMMAND lighttouch_archive_import_test)
 add_executable(lighttouch_nand_disk_test tests/NandDisk.cpp)
 target_link_libraries(lighttouch_nand_disk_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_nand_disk_test PRIVATE -UNDEBUG)
 add_test(NAME lighttouch_nand_disk COMMAND lighttouch_nand_disk_test)
 add_executable(lighttouch_observed_admission_test tests/ObservedAdmission.cpp)
 target_link_libraries(lighttouch_observed_admission_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_observed_admission_test PRIVATE -UNDEBUG)
 add_test(NAME lighttouch_observed_admission COMMAND lighttouch_observed_admission_test)
 add_executable(lighttouch_pcm_epoch_test tests/PcmEpoch.cpp)
 target_link_libraries(lighttouch_pcm_epoch_test PRIVATE Threads::Threads)
 target_compile_options(lighttouch_pcm_epoch_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_pcm_epoch COMMAND lighttouch_pcm_epoch_test)
 add_executable(lighttouch_shutdown_admission_test tests/GuestShutdownAdmission.cpp)
 target_compile_options(lighttouch_shutdown_admission_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_shutdown_admission COMMAND lighttouch_shutdown_admission_test)
 add_executable(lighttouch_guest_package_judge_test tests/GuestPackageJudge.cpp)
 target_compile_options(lighttouch_guest_package_judge_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_guest_package_judge COMMAND lighttouch_guest_package_judge_test)
 add_executable(lighttouch_frame_storage_test tests/FrameStorage.cpp)
 target_include_directories(lighttouch_frame_storage_test PRIVATE include)
 target_link_libraries(lighttouch_frame_storage_test PRIVATE Threads::Threads)
 target_compile_options(lighttouch_frame_storage_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_frame_storage COMMAND lighttouch_frame_storage_test)
 # Include the production entry points in this private fixture TU to exercise
 # the opaque runtime without an engine stub or a public test-only API.
 add_executable(lighttouch_runtime_frame_metadata_test tests/RuntimeFrameMetadata.cpp)
 target_include_directories(lighttouch_runtime_frame_metadata_test PRIVATE "${LIGHTTOUCH_QEMU_SOURCE}/contrib/ios-app" "${LIGHTTOUCH_QEMU_SOURCE}/contrib/macos-app")
 target_link_libraries(lighttouch_runtime_frame_metadata_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_runtime_frame_metadata_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_SYSTEM_NAME STREQUAL "Android")
  target_compile_definitions(lighttouch_runtime_frame_metadata_test PRIVATE LIGHTTOUCH_TEST_WRAP_THROW=1)
  target_link_options(lighttouch_runtime_frame_metadata_test PRIVATE "LINKER:--wrap=__cxa_throw")
 endif()
 if(LIGHTTOUCH_ENABLE_DEBUG_QMP)
  target_compile_definitions(lighttouch_runtime_frame_metadata_test PRIVATE LIGHTTOUCH_ENABLE_DEBUG_QMP=1)
 endif()
 add_test(NAME lighttouch_runtime_frame_metadata COMMAND lighttouch_runtime_frame_metadata_test)
endif()
